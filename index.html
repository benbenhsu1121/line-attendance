  const firebaseConfig = {

// For Firebase JS SDK v7.20.0 and later, measurementId is optional

  apiKey: "AIzaSyCQ98ZWCCOk7ALopTQKjpizcgeyiRu12_I",

  projectId: "payroll-attendance-da6b1",

  storageBucket: "payroll-attendance-da6b1.firebasestorage.app",

  messagingSenderId: "697411530182",

  appId: "1:697411530182:web:82a967282a51148c0016c7",

  measurementId: "G-126PX9RFPE"

};


// Initialize Firebase

const app = initializeApp(firebaseConfig);

const analytics = getAnalytics(app);

  };


  // TODO 2: 填入步驟 2 取得的 LIFF ID

  const MY_LIFF_ID = "2011667192-mzmPgQFd";


  const app = initializeApp(firebaseConfig);

  const db = getFirestore(app);


  let currentCoords = null;

  let currentAddress = "未知地址";

  let currentUser = { userId: "", displayName: "" };


  // 更新時間顯示

  setInterval(() => {

    const now = new Date();

    document.getElementById("currentTime").innerText = now.toLocaleTimeString();

  }, 1000);


  // 初始化 LIFF

  async function initLiff() {

    try {

      await liff.init({ liffId: MY_LIFF_ID });

      if (!liff.isLoggedIn()) {

        liff.login();

        return;

      }

      const profile = await liff.getProfile();

      currentUser.userId = profile.userId;

      currentUser.displayName = profile.displayName;


      document.getElementById("userName").innerText = profile.displayName;

      document.getElementById("userUid").innerText = `ID: ${profile.userId}`;

      if (profile.pictureUrl) {

        document.getElementById("userPicture").src = profile.pictureUrl;

      }


      getLocation();

    } catch (err) {

      document.getElementById("status").innerText = "LINE 授權失敗: " + err.message;

    }

  }


  // 取得 GPS 經緯度

  function getLocation() {

    if (!navigator.geolocation) {

      document.getElementById("status").innerText = "瀏覽器不支援定位";

      return;

    }


    navigator.geolocation.getCurrentPosition(

      async (pos) => {

        currentCoords = {

          latitude: pos.coords.latitude,

          longitude: pos.coords.longitude

        };

        await reverseGeocode(currentCoords.latitude, currentCoords.longitude);

      },

      (err) => {

        document.getElementById("status").innerText = "定位失敗，請確認已開放 GPS 權限";

      },

      { enableHighAccuracy: true, timeout: 10000 }

    );

  }


  // 反查中文地址 (Nominatim API)

  async function reverseGeocode(lat, lon) {

    try {

      document.getElementById("locAddress").innerText = "正在解析地址...";

      const res = await fetch(`https://nominatim.openstreetmap.org/reverse?format=json&lat=${lat}&lon=${lon}&zoom=18&addressdetails=1`, {

        headers: { "Accept-Language": "zh-TW" }

      });

      const data = await res.json();

      currentAddress = data.display_name || `${lat.toFixed(4)}, ${lon.toFixed(4)}`;

      document.getElementById("locAddress").innerText = currentAddress;


      // 啟用打卡按鈕

      document.getElementById("btnIn").disabled = false;

      document.getElementById("btnOut").disabled = false;

      document.getElementById("status").innerText = "準備就緒，請點選打卡";

    } catch {

      currentAddress = `經緯度: (${lat.toFixed(4)}, ${lon.toFixed(4)})`;

      document.getElementById("locAddress").innerText = currentAddress;

      document.getElementById("btnIn").disabled = false;

      document.getElementById("btnOut").disabled = false;

    }

  }


  // 寫入打卡紀錄至 Firestore

  window.submitCheckIn = async function(type) {

    document.getElementById("btnIn").disabled = true;

    document.getElementById("btnOut").disabled = true;

    document.getElementById("status").innerText = "打卡資料傳輸中...";


    try {

      await addDoc(collection(db, "check_ins"), {

        lineUserId: currentUser.userId,

        displayName: currentUser.displayName,

        type: type,

        latitude: currentCoords.latitude,

        longitude: currentCoords.longitude,

        address: currentAddress,

        timestamp: serverTimestamp(),

        syncedToLocal: false

      });


      alert(type === "clock_in" ? "上班打卡成功！" : "下班打卡成功！");

      document.getElementById("status").innerText = "打卡成功，視窗即將關閉...";

      setTimeout(() => { liff.closeWindow(); }, 1500);

    } catch (e) {

      alert("打卡失敗: " + e.message);

      document.getElementById("btnIn").disabled = false;

      document.getElementById("btnOut").disabled = false;

      document.getElementById("status").innerText = "傳輸失敗，請重試";

    }

  };


  initLiff();
